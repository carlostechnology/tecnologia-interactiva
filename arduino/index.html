Input → Output

Arduino is a programmable board. A button is an input. An LED is an output. The program connects them.

08PRACTICE 08 · BUILD & LEARNTime is running!Can Arduino measure time?○⌄

millis() measures time in milliseconds. 1000 ms = 1 second. Unlike a long delay(), it lets our program keep checking buttons.

01 — TRY IT ONLINEBROWSER SIMULATION · NO ARDUINO NEEDED
SECONDS LEFT10
COUNTDOWN
02 — ARDUINO CODEpractice_08.ino
Arduino · C++
const unsigned long DURATION = 10000;

unsigned long startTime;
unsigned long lastUpdate = 0;

void setup() {
  Serial.begin(115200);
  startTime = millis();
}

void loop() {
  unsigned long now = millis();
  unsigned long elapsed = now - startTime;

  unsigned long remaining;

  if (elapsed >= DURATION) {
    remaining = 0;
  } else {
    remaining = DURATION - elapsed;
  }

  if (now - lastUpdate >= 200) {
    lastUpdate = now;
    Serial.println((remaining + 999) / 1000);
  }
}

09PRACTICE 09 · BUILD & LEARNWhose turn is it?Can Arduino remember the active player?○⌄

The chess clock uses states: WAITING, WHITE_TURN and BLACK_TURN. Press BLACK first to start White’s clock.

01 — TRY IT ONLINEBROWSER SIMULATION · NO ARDUINO NEEDED
ACTIVE PLAYERWAITINGPress BLACK first
02 — ARDUINO CODEpractice_09.ino
Arduino · C++
const int WHITE_BUTTON = A0;
const int BLACK_BUTTON = A3;

enum State {
  WAITING,
  WHITE_TURN,
  BLACK_TURN
};

State state = WAITING;

bool oldWhite = false;
bool oldBlack = false;

void setup() {
  pinMode(WHITE_BUTTON, INPUT);
  pinMode(BLACK_BUTTON, INPUT);

  Serial.begin(115200);
  Serial.println("WAITING");
}

void loop() {
  bool white = digitalRead(WHITE_BUTTON) == HIGH;
  bool black = digitalRead(BLACK_BUTTON) == HIGH;

  if (black && !oldBlack) {
    if (state == WAITING || state == BLACK_TURN) {
      state = WHITE_TURN;
      Serial.println("WHITE TURN");
    }
  }

  if (white && !oldWhite) {
    if (state == WHITE_TURN) {
      state = BLACK_TURN;
      Serial.println("BLACK TURN");
    }
  }

  oldWhite = white;
  oldBlack = black;

  delay(20);
}

10PRACTICE 10 · FINAL PROJECTBuild your chess clock!It's time for the final project!○⌄

Two players, five minutes each, two buttons, one LED, one buzzer and one OLED. During the game the LED stays on. At zero, the alarm sounds and the LED blinks. Hold both buttons for two seconds to reset.

01 — TRY IT ONLINEBROWSER SIMULATION · NO ARDUINO NEEDED

00:10
LED
BLACK
00:10
Press BLACK to start

Browser simulation: 10 seconds per player for a quick test. On Arduino, the program starts at 5 minutes. The physical version resets by holding both buttons for 2 seconds.

02 — ARDUINO CODEpractice_10.ino
Arduino · C++
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// ---------- PINS ----------
const int WHITE_BUTTON = A0;
const int BLACK_BUTTON = A3;
const int LED = A1;
const int BUZZER = A2;

Adafruit_SSD1306 oled(128, 64, &Wire, -1);

// ---------- TIME ----------
const unsigned long GAME_TIME = 5UL * 60UL * 1000UL;
const unsigned long RESET_TIME = 2000UL;

unsigned long whiteTime = GAME_TIME;
unsigned long blackTime = GAME_TIME;

unsigned long lastTime = 0;
unsigned long lastScreen = 0;
unsigned long holdStart = 0;

// ---------- STATES ----------
enum State {
  WAITING,
  WHITE_TURN,
  BLACK_TURN,
  FINISHED
};
State state = WAITING;

// ---------- BUTTONS ----------
bool oldWhite = false;
bool oldBlack = false;
bool bothBefore = false;
bool buttonsLocked = false;
bool resetDone = false;

// ---------- SOUNDS ----------
void beep() {
  tone(BUZZER, 1800, 60);
}

void alarmSound() {
  for (int i = 0; i < 3; i++) {
    tone(BUZZER, 2000, 200);
    delay(300);
  }
}

// ---------- DISPLAY TIME ----------
void showTime(unsigned long ms) {
  unsigned long seconds = (ms + 999UL) / 1000UL;
  int minutes = seconds / 60;
  int rest = seconds % 60;

  if (minutes < 10) oled.print("0");
  oled.print(minutes);
  oled.print(":");

  if (rest < 10) oled.print("0");
  oled.print(rest);
}

// ---------- OLED ----------
void showScreen() {
  oled.clearDisplay();
  oled.setTextColor(SSD1306_WHITE);

  oled.setTextSize(1);
  oled.setCursor(8, 0);
  oled.print("WHITE");
  oled.setCursor(76, 0);
  oled.print("BLACK");

  oled.setTextSize(2);
  oled.setCursor(2, 18);
  showTime(whiteTime);
  oled.setCursor(68, 18);
  showTime(blackTime);

  oled.drawLine(0, 42, 127, 42, SSD1306_WHITE);
  oled.setTextSize(1);
  oled.setCursor(4, 50);

  if (state == WAITING) {
    oled.print("Press BLACK");
  }
  else if (state == WHITE_TURN) {
    oled.print("WHITE TURN");
  }
  else if (state == BLACK_TURN) {
    oled.print("BLACK TURN");
  }
  else if (whiteTime == 0) {
    oled.print("BLACK WINS");
  }
  else {
    oled.print("WHITE WINS");
  }
  oled.display();
}

// ---------- RESET ----------
void resetClock() {
  whiteTime = GAME_TIME;
  blackTime = GAME_TIME;
  state = WAITING;
  noTone(BUZZER);
  digitalWrite(LED, LOW);
  lastTime = millis();
  showScreen();
}

// ---------- SETUP ----------
void setup() {
  pinMode(WHITE_BUTTON, INPUT);
  pinMode(BLACK_BUTTON, INPUT);
  pinMode(LED, OUTPUT);
  pinMode(BUZZER, OUTPUT);
  Wire.begin();

  if (!oled.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    while (true) {
      delay(100);
    }
  }
  resetClock();
}

// ---------- LOOP ----------
void loop() {
  unsigned long now = millis();
  unsigned long elapsed = now - lastTime;
  lastTime = now;

  // 1. Update the active timer
  if (state == WHITE_TURN) {
    if (elapsed >= whiteTime) {
      whiteTime = 0;
      state = FINISHED;
      alarmSound();
    } else {
      whiteTime -= elapsed;
    }
  }
  else if (state == BLACK_TURN) {
    if (elapsed >= blackTime) {
      blackTime = 0;
      state = FINISHED;
      alarmSound();
    } else {
      blackTime -= elapsed;
    }
  }

  // 2. Read the buttons
  bool white = digitalRead(WHITE_BUTTON) == HIGH;
  bool black = digitalRead(BLACK_BUTTON) == HIGH;

  // 3. Hold both buttons to reset
  if (white && black) {
    buttonsLocked = true;
    if (!bothBefore) {
      holdStart = now;
    }
    bothBefore = true;
    if (!resetDone && now - holdStart >= RESET_TIME) {
      resetClock();
      resetDone = true;
    }
  }
  else {
    bothBefore = false;
    if (!white && !black) {
      buttonsLocked = false;
      resetDone = false;
    }

    // 4. Change turns
    if (!buttonsLocked && state != FINISHED) {
      if (state == WHITE_TURN && white && !oldWhite) {
        state = BLACK_TURN;
        beep();
      }
      if ((state == WAITING || state == BLACK_TURN) &&
          black && !oldBlack) {
        state = WHITE_TURN;
        beep();
      }
    }
  }

  oldWhite = white;
  oldBlack = black;

  // 5. Red LED
  if (state == FINISHED) {
    digitalWrite(LED, (millis() / 250) % 2);
  } else {
    bool playing = state == WHITE_TURN || state == BLACK_TURN;
    digitalWrite(LED, playing);
  }

  // 6. Update the screen
  if (millis() - lastScreen >= 100) {
    lastScreen = millis();
    showScreen();
  }
  delay(20);
}
